\chapter{От микроуровня к макроописанию}
\label{ch:macro}

\section{Необходимость перехода}

Планковские $\hl$ и ячейка~$\PlanckCell$ определены через $\hbar$, $G$, $c$,
но принципиально недоступны непосредственному измерению: слишком малы масштаб и объём.
Классическая физика часто берёт предел $c\to\infty$ или $\hl\to 0$;
на микроуровне~$M$ такой предел отвергнут (\cref{sec:no-continuum-carrier}).

Макроскопический мир~$T$ --- единственный слой прямого доступа наблюдателя.
Из~$M$ \emph{обязано} следовать~$T$: осреднение поля, дискретный анализ, статистика измерений.
Согласование модели с экспериментом проверяется на~$T$ (дисперсия, массы, Стандартная модель);
микроуровень подтверждается косвенно: если макроинварианты сходятся,
каркас не опровергнут.

\section{Электрон как составная структура}

Центральный вопрос перехода $M\to T$: из чего на микроуровне состоит
макроскопический электрон с известными массой~$m_e$ и зарядом~$e$?
Предел масштаба --- ячейка~$\PlanckCell$, а не составная частица меньшего размера.
Проверка теории --- способность комбинаций ячеек и закона~$g$
воспроизвести электрон без новых подгоняемых параметров.

\section{Оператор осреднения \texorpdfstring{$\mathcal{B}$}{B}}
\label{sec:readout}

\begin{definition}[Осреднение с биномиальным ядром]
\label{def:soft-readout}
На ортогональном срезе произведения осей
\[
  \Phi(X,t)=\bigl(\tfrac14[1,2,1]\bigr)^{\otimes 2R} * z(x,t).
\]
Эквивалентно одному проходу свёртки $3\times 3$ с ядром $[1,2,1;2,4,2;1,2,1]/16$.
\end{definition}

За один такт~$\htick$ центральный узел не лежит в светоподобной оболочке (условие~$A_1$).
Минимальная чётная глубина дискретного «зелёного'' ядра --- два такта (туда и обратно):
на цепи~$\mathbb{Z}$ это.
\[
[1,1]/2*[1,1]/2=[1,2,1]/4
\]
На FCC полное осреднение задаётся суммированием по путям длины~2 на графе~$N_{12}$,
а не разложимой свёрткой.
\[
(1{:}2{:}1)^{\otimes 3}
\]

\section{Теорема T-CR: континуум как следствие осреднения}
\label{sec:t-cr}

\begin{theorem}[T-CR]
\label{thm:t-cr}
Континуум не постулат микроуровня~$M$.
При фиксированной $\ell_P$ и радиусе осреднения $R\ge 1$
любой прибор с ядром~$\mathcal{B}$ (условия C0--C4) видит поле,
неотличимое от непрерывного на шкалах $\lambda\gg\sigma_R\ell_P$,
с относительной ошибкой $O((\sigma_R\ell_P/\lambda)^4)$.
\end{theorem}

\begin{proof}
По \cref{def:soft-readout} ядро осреднения~$\mathcal{B}$ --- $R$-кратная свёртка
с $W=\tfrac14[1,2,1]^{\otimes 2R}$ на ортогональном срезе (условия C0--C4:
локальность, нормировка $\sum W=1$, чётность моментов, separable на product-срезе).

\smallskip
\emph{Одномерная цепь.}
Для одного прохода $W=[1,2,1]/4$ символ Фурье
$\widehat W(k)=\tfrac12+\tfrac12\cos k=\cos^2(k/2)$.
После $R$ проходов $\widehat W_R(k)=\cos^{2R}(k/2)$, дисперсия $\sigma_R^2=R/2$.
Разложение $\ln\cos(k/2)=-k^2/8-k^4/96+O(k^6)$ даёт
\[
  \frac{\widehat W_R(k)}{G_R(k)}=\exp\Big(-\frac{Rk^4}{96}+O(Rk^6)\Big),
  \quad G_R(k)=\exp(-Rk^2/4).
\]
При $k\sigma_R\ll 1$ относительная ошибка замены $W_R$ гауссовым ядром --- $O((k\sigma_R)^4)$.
Физическая длина волны $\lambda=2\pi/k$ при $\ell_P=\hl$ даёт
$O((\sigma_R\ell_P/\lambda)^4)$ --- формулировка теоремы.

\smallskip
\emph{Ультрафиолет и отсутствие $\hl\to 0$.}
При $k\sim 1$ (масштаб $\ell_P$) имеем $\widehat W_R(k)\le e^{-cR}$: осреднение подавляет
УФ-моды без предела $\hl\to 0$; континуум --- свойство readout'а на~$T$, не аксиома~$M$.

\smallskip
\emph{FCC в $(3{+}1)$.}
На FCC ядро depth-2 path-Green (\cref{def:soft-readout}, гл.~\ref{ch:macro})
даёт $w(0)=1/12$ и изотropный второй момент $M_{ij}=\tfrac43\delta_{ij}$
(полное перечисление двухшаговых путей на~$N_{12}$).
Separable накопление $W^{\otimes 2R}$ переносит одномерную оценку $O(k^4)$
на каждую ось product-среза; изотropия $M$ обеспечивает одну шкалу $\sigma_R$ во всех
направлениях readout'а.
\end{proof}

\section{Свободная энергия Ginzburg--Landau}
\label{sec:gl-free-energy}

\paragraph{Объект.}
На~$\Tlayer$ прибор не видит микроскопическое $z(x,t)$ поштучно: после осреднения~\cref{def:soft-readout}
остаётся медленное поле $\Phi=\mathcal{B}z$ на объёме~$V$.
Его эволюцию на макрошкалах ($\lambda\gg\sigma_R\ell_P$, \cref{thm:t-cr}) описывают
\emph{эффективной} свободной энергией --- как параметр порядка в теории Ландау
(ср.~Landau--Lifshitz, т.~IX, \S45--47; здесь роль $\psi$ играет~$\Phi$).

\paragraph{Разложение.}
Вне плотных вихрей вращатель $R(\Phi)$ разложим по малой плотности $\rho_E\ll K_P$
(\cref{def:kp}, гл.~\ref{ch:evolution}): saturating gate даёт $\sim|\Phi|^4$,
осреднение фазовых градиентов --- $\sim|\nabla\Phi|^2$
(гл.~\ref{ch:evolution}, \ref{ch:matter}).
По схеме Ландау для второго рода:
\begin{equation}
\label{eq:gl-free-energy}
  \mathcal{F}_{\mathrm{GL}}[\Phi]
  = \mathcal{F}_0
  + \int_V \Bigl(
      \alpha\,|\Phi|^2
      + \frac{\beta}{2}\,|\Phi|^4
      + \frac{\hbar^2}{2m}\,|\nabla\Phi|^2
    \Bigr)\,\mathrm{d}^3x.
\end{equation}
$\mathcal{F}_0$ --- энергия однородного фона; $\alpha\,|\Phi|^2$ --- квадратичный член;
$\beta>0$ --- насыщение при $\rho_E\to K_P$; в замыкании $\beta\equiv\omega_{\mathrm{macro}}$.

\paragraph{Вариация.}
При фиксированных граничных данных на $\partial V$:
\begin{equation}
\label{eq:gl-variation}
  \frac{\delta\mathcal{F}_{\mathrm{GL}}}{\delta\Phi^*}
  = \alpha\,\Phi
  - \frac{\hbar^2}{2m}\,\nabla^2\Phi
  + \beta\,|\Phi|^2\Phi.
\end{equation}
Условие $\delta\mathcal{F}_{\mathrm{GL}}=0$ --- статика поля на~$\Tlayer$.

\paragraph{Динамика.}
Для \emph{медленной} эволюции $\Phi(t)$ вводим уравнение типа time-dependent Ginzburg--Landau
(аналогично связи гамильтониана и уравнения Шрёдингера: Фейнман, лекции, т.~III, гл.~5--6):
\begin{equation}
\label{eq:gl-dynamics}
  i\hbar\,\frac{\partial\Phi}{\partial t}
  = \frac{\delta\mathcal{F}_{\mathrm{GL}}}{\delta\Phi^*}.
\end{equation}
Это не новый постулат на~$M$: на микроуровне по-прежнему действует только $g$;
\eqref{eq:gl-dynamics} --- сжатое описание того, как прибор видит медленную компоненту после~$\mathcal{B}$.

\section{Нелинейное уравнение Шрёдингера и гидродинамика}
\label{sec:nlse-hydro}

\paragraph{Вывод.}
Подставляем~\eqref{eq:gl-variation} в~\eqref{eq:gl-dynamics}:
\begin{equation}
\label{eq:nlse-gl}
  i\hbar\,\frac{\partial\Phi}{\partial t}
  = \alpha\,\Phi
  - \frac{\hbar^2}{2m}\,\nabla^2\Phi
  + \beta\,|\Phi|^2\Phi.
\end{equation}
Это нелинейное уравнение Шрёдингера типа Ginzburg--Landau.
В координатах плотности $|\Phi|^2$, при $\alpha=0$ на выбранном фоне, оно эквивалентно записи
\begin{equation}
\label{eq:nlse-gl-real}
  \frac{\partial\Phi}{\partial t}
  = \frac{i\hbar}{2m}\,\nabla^2\Phi
  - i\,\beta\,|\Phi|^2\Phi.
\end{equation}

\paragraph{Диссипация readout'а.}
Член $\nu_{\mathrm{CA}}\nabla^2\Phi$ \emph{не} следует из~\eqref{eq:gl-free-energy}:
он появляется только после учёта необратимости~$\mathcal{B}$ на~$\Tlayer$
($\Axref{ax:A6}$, \cref{lem:236}, \cref{prop:237}) и дискретного coarse-graining (\cref{sec:t-cr}).
$\mathcal{F}_{\mathrm{GL}}$ задаёт \emph{обратимую} часть; $\nu_{\mathrm{CA}}$ --- кинематическая вязкость
макроуровня (ср.~диффузионное дополнение к волновому уравнению в курсе полей).
Полное уравнение на~$\Tlayer$:
\[
  \frac{\partial\Phi}{\partial t}
  = \frac{i\hbar}{2m}\,\nabla^2\Phi
  - i\,\beta\,|\Phi|^2\Phi
  + \nu_{\mathrm{CA}}\,\nabla^2\Phi,
\]
новый оператор на~$M$ при этом не вводится.
