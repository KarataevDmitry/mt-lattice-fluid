\section{Иерархия \texorpdfstring{$\Mlayer\to\mathrm{SM}$}{M→SM}}

Лестница масштабов: $\hv$ $\to$ pra-дефект $\to$ составные узлы $\to$ $\Tlayer$-частицы $\to$ SM.
Каждый уровень --- readout предыдущего без новых knobs в $g$.

\section{SI-мост}

\begin{definition}[Planck-идентификация]
$\ell_P:=\hl$, $a=\hl$, $\kgeom$ из 1-tick hull (\cref{ch:carrier}):
\[
  \htick=\kgeom\, t_P,\quad E_0=\kgeom E_P,\quad c=\kgeom c_0.
\]
\end{definition}

\begin{remark}
Planck $t_P=\ell_P/c$ уже несёт macro-$c$; $\kgeom$ не определяется как $c/c_0$,
а выводится из геометрии, затем $c=\kgeom c_0$ --- проверка.
\end{remark}

\section{Тонкая структура}

\begin{theorem}[$\alpha_{\mathrm{fs}}$ из gate]
\label{thm:alpha}
\[
  \alpha_{\mathrm{fs}}^{-1}=4\pi^3+\pi^2+\pi\approx 137.036,
\]
не combinatorics $|N|\pi^3+\cdots$.
\end{theorem}

\begin{proof}[Идея]
$\alpha^*-1=\Delta\phi_{\min}/(2\pi)=1/(4\pi)$ из вакуумного gate.
$\alpha_{\mathrm{fs}}$ --- безразмерное сопряжение фазовых каналов на $\Tlayer$;
полный вывод через holonomy $\Phi_\square$ и readout Maxwell (\cref{ch:carrier}).
\end{proof}

\section{Электрослабый сектор}

Пиковая плотностная шкала (\cref{sec:theorem-51}, bit budget):
\[
  N_{\mathrm{hier}}=\lfloor B_{\hv}\rfloor-1=8,\qquad
  v_{\mathrm{peak}}=\alpha_{\mathrm{fs}}^{8} E_P.
\]

\begin{theorem}[VEV из readout]
\label{thm:vev}
\[
  v=v_{\mathrm{peak}}\sqrt{2\pi}=\alpha_{\mathrm{fs}}^{8} E_P\sqrt{2\pi}\approx 246\ \mathrm{GeV}.
\]
\end{theorem}

\begin{proof}
$N_{\mathrm{hier}}$ --- бит иерархии минус бит занятости ячейки.
Множитель $\sqrt{2\pi}$ --- Gaussian-мера binomial readout (\cref{thm:t-cr}):
пик плотности vs интегральный порядковый параметр на $\Tlayer$.
\end{proof}

\section{Четыре взаимодействия}

\begin{proposition}[Один $g$ --- четыре канала readout]
Электромагнетизм: фазы на рёбрах, $\Phi_\square$, $n=0$ излучение.
Слабое: $P_L/P_R$, SU(2) flip; $\sin^2\theta_W=3/13$ из геометрии кластера $12+1$.
Сильное: составные узлы, $\alpha_s(v)=d/(N_{\mathrm{hier}}\pi)$.
Гравитация: $\Tlayer$-strain, $G_{\mu\nu}=\mathcal{E}(\delta[\mathrm{strain}])=8\pi\ell_P^2 T_{\mu\nu}$.
\end{proposition}

Открытые выводы (stencil GW, полный census барионов, CKM-углы) --- в \texttt{DEVLOG.md};
они не ослабляют формулировки теорем выше.
