\chapter{Дискретность микроуровня}
\label{ch:foundations}

\section{От постулата к теореме}

Ранний постулат «решётка существует'' снят.
При абсолютных условиях (гл.~\ref{ch:axioms}) и геометрии носителя (гл.~\ref{ch:carrier})
дискретность~$\Lambda$, шаг~$\hl$ и такт~$\htick$ \emph{выводятся}.

\begin{lemma}[Счётное пространство конфигураций]
\label{lem:countable-config}
При \cref{def:field-a3} пространство микросостояний
\[
  \mathcal{C}=\bigl\{z(x)\in\mathbb{C}^2\bigr\}_{x\in\Lambda}
\]
с дискретным алфавитом на каждом узле счётно, а эволюция $g:\mathcal{C}\to\mathcal{C}$ --- биекция.
\end{lemma}

\begin{proof}
На узле~$x$ состояние принимается из конечного кольца $\mathbb{Z}_{N_{\mathrm{ring}}}[i]$;
мощность кольца фиксируется информационным бюджетом~$B_{\hv}$ (\cref{sec:bit-budget}).
При счётном~$\Lambda$ множество $\mathcal{C}$ есть счётное произведение конечных множеств, значит, $\mathcal{C}$ счётно.
Норма сохраняется, и у шага есть обратный:
$g$ --- перестановка (биекция) на~$\mathcal{C}$.
\end{proof}

\begin{lemma}[FCC-носитель и элементарная шкала]
\label{lem:fcc-scale}
Носитель --- $\Lambda=\mathrm{FCC}$ с $|N(x)|=12$ (\cref{thm:fcc-choice}).
Шаг~$\hl$ и такт~$\htick$ однозначно задаются однотактовым телом FCC (\cref{sec:fcc-kappa});
$\hv$ --- узел решётки (регистр~\cref{def:field-a3}).
\end{lemma}

\begin{proof}
Теорема~\ref{thm:fcc-choice} фиксирует $\Lambda=\mathrm{FCC}$ и $|N|=12$.

\smallskip
\emph{Шаг~$\hl$.}
Пусть $a=\hl$ --- ребро кубооктаэдра однотактового тела (\cref{sec:fcc-kappa}).
Объём ячейки Вороного FCC при $a=\hl$ равен $v_{\hv}=a^3/\sqrt{2}=\hl^3/\sqrt{2}$;
это минимальный регистр, согласованный с плотнейшей упаковкой.

\smallskip
\emph{Шаг~$\htick$.}
Тот же раздел задаёт $\kgeom=R_{\mathrm{in}}/R_{\mathrm{out}}=1/\sqrt{2}$ и
$c_0=\hl/\htick$.
После SI-моста (\cref{def:si-bridge}) $\htick=\kgeom\,t_P$ при $\ell_P:=\hl$,
откуда $c_0=\sqrt{2}\,c$ --- согласованность тактовой и макроскопической скоростей.

\smallskip
\emph{Ячейка~$\hv$.}
По \cref{def:field-a3} $\PlanckCell$ --- узел FCC (место регистра~$z$); $v_{\hv}$ --- объём Вороного этого узла.
\end{proof}

\begin{lemma}[Атомность регистра]
\label{lem:register-atom}
Элементарная ячейка~$\hv$ не дробится
на под-решётку с шагом $\hl'<\hl$ без смены алфавита и закона~$g$.
\end{lemma}

\begin{proof}
Два совпадающих спинора в одном~$\PlanckCell$ исключены.
Минимальный фазовый шаг
\[
  \Delta\phi_{\min}=\tfrac12,
\]
откуда
\[
  s_0=\hbar/2,\qquad p_0=\frac{\hbar}{2\hl}.
\]
Топологический заряд $n\in\mathbb{Z}$ и полюс сидят на границе одной ячейки~$\hv$.
Размазывание полюса по подузлам разнесло бы этот заряд.
Потолок плотности и нижняя граница $|z|\ge z_{\min}>0$ фиксируют бюджет~$B_{\hv}$ на регистр.
Формальное дробление~$\hv$ означало бы несколько независимых алфавитов на месте одного узла FCC, то есть другой закон.
\end{proof}

\begin{theorem}[Дискретность и минимальная ячейка $\hv$]
\label{thm:discreteness}
Пусть $g$ --- допустимый локальный закон на~$M$,
удовлетворяющий абсолютным условиям~\cref{ch:axioms} на геометрии FCC-носителя~\cref{ch:carrier}.
Тогда:
\begin{enumerate}[label=(\roman*)]
\item $g$ --- дискретная динамика на~$\Lambda$;
\item существуют счётный FCC-носитель~$\Lambda$, элементарная ячейка~$\hv$, шаг~$\hl$ и такт~$\htick$;
\item дробление регистра «ниже~$\hl$'' меняет алфавит: под-решётка несёт другой закон.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{(i)}~Предположим, что $g$ --- «поле на континууме'' в классическом смысле:
несчётное семейство $z(x)$ при $x\in\mathbb{R}^3$ с непрерывными координатами и произвольными
значениями в~$\mathbb{C}^2$.
Тогда уже одна точка $x$ несёт несчётное число степеней свободы, тогда как допустимый закон
действует на счётном пространстве конфигураций (\cref{lem:countable-config}) с конечным алфавитом
на узле.
Биективная эволюция~$g$ на таком пространстве не воспроизводит континуальный поток:
у потока на~$\mathbb{R}^3$ степеней свободы больше, чем у конечного алфавита на узле.
Следовательно, $g$ --- дискретная динамика на~$\Lambda$.

\smallskip
\emph{(ii)}~Носитель и шкала уже выведены.
Теорема~\ref{thm:fcc-choice} даёт $\Lambda=\mathrm{FCC}$ и $|N|=12$.
Объём ячейки Вороного при ребре $a=\hl$ равен $v_{\hv}=\hl^3/\sqrt{2}$ (\cref{sec:fcc-kappa}).
Отношение радиусов оболочки $\kgeom=1/\sqrt{2}$ фиксирует такт через макроскопическую скорость: $\htick=\kgeom\, t_P$ при $\ell_P:=\hl$, откуда $c_0=\sqrt{2}\,c$.
Узел регистра --- одна ячейка этой решётки (\cref{def:field-a3}).

\smallskip
\emph{(iii)}~Если бы подрешётка с шагом $\hl'<\hl$ несла тот же закон~$g$,
на одном узле FCC оказалось бы несколько регистров с тем же топологическим зарядом на границе одной ячейки.
Это запрещено \cref{lem:register-atom}: заряд и устойчивость полюса сидят на границе именно одной~$\hv$, а бюджет регистра не делится без смены алфавита.
\end{proof}

\begin{corollary}[Дискретность как следствие]
\label{cor:discreteness}
Дискретность микроуровня есть содержание теоремы~\ref{thm:discreteness}.
\end{corollary}

\section{Состояние, континуум, вакуум}

\begin{corollary}[Континуум на~$M$ отсутствует]
\label{cor:no-continuum}
Непрерывные поля и предел $\hl\to 0$ --- свойство макроуровня~$T$
(гл.~\ref{sec:no-continuum-carrier}).
\end{corollary}

\begin{corollary}[Нижняя граница амплитуды]
\label{cor:vacuum-floor}
$|z(x)|\ge z_{\min}>0$ на~$M$.
\end{corollary}

\section{Замыкание закона, механика, кванты}

\begin{corollary}[Замыкание $g$]
\label{cor:g-closure}
Допустимый закон~$g$ --- закон теоремы~\cref{thm:g-law}.
\end{corollary}

\begin{corollary}[Лестница Ландау]
\label{cor:landau}
представление Маделунга --- язык~$T$.
\end{corollary}

\begin{corollary}[Дискретные возбуждения]
\label{cor:full-quantization}
На~$M$ --- занятость мод $n_k\in\mathbb{Z}$; «волны'' --- макроскопическое осреднение на~$T$.
\end{corollary}
