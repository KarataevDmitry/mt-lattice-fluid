\section{От постулата к теореме}

\begin{remark}[Постулат 0.1]
Ранний постулат «решётка существует'' снят.
При абсолютных условиях \cref{ch:axioms} и геометрии носителя \cref{ch:carrier}
дискретность $\Lambda$, шаг $\hl$ и тик $\htick$ \emph{выводятся} (\cref{thm:discreteness}),
а не вводятся отдельной аксиомой «CA exists''.
\end{remark}

\begin{theorem}[Дискретность и минимальная ячейка $\hv$]
\label{thm:discreteness}
Пусть $g$ --- допустимый локальный закон, согласованный с \cref{ch:axioms,ch:carrier,sec:g-law}.
Тогда:
\begin{enumerate}[label=(\roman*)]
\item $g$ не есть поле на континууме $x\in\mathbb{R}^3$;
\item существует счётный FCC-носитель $\Lambda$, ячейка $\hv$, шаг $\hl$, тик $\htick$;
\item под-решётка «ниже $\hl$'' не состояние $\Mlayer$ --- запрет дробления регистра.
\end{enumerate}
\end{theorem}

\begin{proof}
\paragraph{Шаг 1.}
\cref{ax:A3,ax:A13}: алфавит $Z_N[i]$ на сайте конечен; при счётном $\Lambda$ конфигурации счётны;
$g$ --- перестановка, не континуумный поток. Пункт (i).

\paragraph{Шаг 2.}
\cref{ax:A1,ax:A2} и \cref{thm:fcc-choice}: $\Lambda=\mathrm{FCC}$, $|N|=12$, $\hl$ и $\htick$ из 1-tick hull
(\cref{ch:carrier}). Пункт (ii).

\paragraph{Шаг 3.}
\cref{ax:A10,ax:A11}: заряд $n\in\mathbb{Z}$ на $\partial(\hv)$; soliton не дробит $\hv$. Пункт (ii), атомность.

\paragraph{Шаг 4.}
\cref{ax:A5,ax:A7}: $z_{\min}>0$, $B_{\hv}=2\pi/\ln 2$ (\cref{sec:bit-budget}).
Дробление $\hv$ --- новый алфавит, не тот же $g$. Пункт (iii).

\paragraph{Шаг 5.}
\cref{ax:A16}: $s_0=\hbar/2$ $\Rightarrow$ $p_0=\hbar/(2\hl)$ фиксирует $\hl$ при данном $\hbar$.
\end{proof}

\begin{corollary}[Дискретность --- следствие]
\label{cor:discreteness}
Дискретность --- следствие \cref{ax:A1,ax:A2,ax:A3,ax:A5,ax:A7,ax:A10,ax:A11,ax:A13,ax:A16},
не отдельный postulate.
\end{corollary}

\section{Состояние, континуум, вакуум}

\begin{corollary}[Континуум на $\Mlayer$ отсутствует]
\label{cor:no-continuum}
Непрерывные поля и $\hl\to 0$ --- только $\Tlayer$ (\cref{sec:no-continuum-carrier}).
\end{corollary}

\begin{corollary}[Планковский пол амплитуды]
\label{cor:vacuum-floor}
$|z(x)|\ge z_{\min}>0$ на $\Mlayer$; A5 (\cref{ax:A5}).
\end{corollary}
