\chapter{Абсолютные условия \texorpdfstring{$A_1$--$A_{16}$}{A1-A16}}
\label{ch:axioms}

\section{Введение}

В этой главе формулируются шестнадцать абсолютных условий на допустимый локальный закон~$g$.
Физическое обоснование каждой группы дано в гл.~\ref{ch:axiom-rationale};
здесь --- строгий реестр с определениями и уравнениями.
Явный вид закона~$g$ выводится в гл.~\ref{ch:evolution} как замыкание этих условий.

\begin{definition}[Поле и норма]
\label{def:field-a3}
\textbf{Носитель.}
$\Lambda$ --- счётная гранецентрированная кубическая (FCC) решётка узлов~$x$
(геометрия --- гл.~\ref{ch:carrier}).
Элементарная ячейка~$\PlanckCell$ --- \emph{один такой узел} (масштаб длины~$\hl$),
а не «точка континуума'' и не макроскопическая частица.
Объём ячейки Вороного записывают отдельно ($v_{\hv}=\hl^3/\sqrt{2}$); символ~$v_p$
в обозначении~$\PlanckCell$ относится к \emph{месту} на решётке, куда записан регистр состояния.

\textbf{Спинор на ячейке.}
На каждом~$x$ задан \emph{спинор}
\[
  z(x,t)=(z_1,z_2)^\top\in\mathbb{C}^2,
\]
то есть столбец из двух комплексных чисел, а не одно число $z\in\mathbb{C}$.
Две компоненты нужны, чтобы микроуровень~$M$ поддерживал заряд ($U(1)$-фаза),
локальный поворот ($SU(2)$) и спин~$1/2$ (оборот $2\pi$ меняет знак состояния; условия~$A_4$, $A_{16}$).
Модуль $|z_1|^2+|z_2|^2$ --- «плотность поля'' на ячейке; фазы $z_k/|z_k|$ --- степени свободы, которые перемешиваются при эволюции.

\textbf{Глобальное состояние.}
Конфигурация всей решётки в такт~$t$:
\[
  \Psi^t=\{z(x,t)\}_{x\in\Lambda}.
\]

\textbf{Норма.}
Суммарная «информационная'' норма (аналог $\braket{\psi|\psi}$ в квантовой механике):
\[
  N(t):=\sum_{x\in\Lambda} z^\dagger(x,t)\,z(x,t)=\sum_x\bigl(|z_1|^2+|z_2|^2\bigr).
\]
Её сохранение при $t\to t+1$ --- содержание условия~$A_3$.
\end{definition}

\section{Каузальность, локальность, унитарность}

В этой секции $\hl$ --- элементарный шаг по решётке (в гл.~\ref{ch:si-sm} отождествляется с~$\ell_P$),
$\htick$ --- один дискретный такт, $c_0=\hl/\htick$ --- тактовая скорость сигнала на~$M$
(не путать с макроскопической~$c$), $N(x)$ --- множество соседей, достижимых из~$x$ за один такт.
Условие~$A_1$ формулирует только каузальность: за $\htick$ учитываются лишь светоподобные рёбра
первой оболочки; включение второй оболочки за такт (на квадратной решётке --- диагональ Moore)
даёт сверхсветовой шаблон и исключается уже этой формулой.
Вывод $|N(x)|=12$ и FCC-носителя, а также соглашение $\ell_P:=\hl$ к~$A_1$ не относятся
(гл.~\ref{ch:carrier}, \cref{thm:fcc-choice}; гл.~\ref{ch:si-sm}, \cref{def:si-bridge}).

\begin{axiom}[A1 · Каузальность]
\label{ax:A1}
За один такт~$\htick$ информация распространяется не дальше одного шага~$\hl$ по решётке.
Локальное обновление использует только \emph{светоподобные} рёбра первой координационной оболочки:
\[
  N(x)=\Bigl\{y:\ |y-x|=\hl,\ \ c_0^2\htick^2-|y-x|^2\ge 0\Bigr\}.
\]
Включение второй оболочки за один такт (на квадратной решётке --- диагональ Moore,
«диагональ за тик'') давало бы сверхсветовой шаблон и запрещено.
\end{axiom}

\begin{axiom}[A2 · Локальность]
\label{ax:A2}
Закон $g$ локален: новое значение $z(x,t+1)$ зависит только от поля в $\varepsilon$-окрестности $x$:
\[
  z(x,t+1)=g\bigl(\{z(y,t)\}_{y\in N(x)}\bigr).
\]
\end{axiom}

\begin{axiom}[A3 · Унитарность / сохранение информации]
\label{ax:A3}
Норма $N(t)$ --- инвариант эволюции:
\[
N(t+1)=N(t)\quad\text{для всех }t.
\]
Линейная часть $g_{\mathrm{lin}}$ --- произведение локальных унитарных операторов на рёбрах $N(x)$.
Нелинейная часть --- только унитарная фаза $z\mapsto z\,e^{i\Phi(z)}$ на $\mathbb{C}^2$; модуль $|z|$ не диссипирует.
\end{axiom}

\begin{axiom}[A4 · U(1) заряд и SU(2) спин]
\label{ax:A4}
Состояние --- спинор $z\in\mathbb{C}^2$, не скаляр $z\in\mathbb{C}$.
Глобальная U(1)-фаза $z\to z\,e^{i\theta}$ --- симметрия вакуума.
Локальное столкновение реализуется SU(2)-вращатель $R(\Phi)$ на обеих компонентах (\cref{sec:g-law}).
\end{axiom}

\section{Термодинамика на ячейке}

\begin{axiom}[A5 · Третье начало (Нернст--Планк)]
\label{ax:A5}
Третье начало термодинамики: при $T\to 0$ энтропия равновесной системы стремится к конечному
пределу (Нернст); в соглашении Планка
\[
  S \xrightarrow[T\to 0]{} 0,
\]
что согласуется со статистикой $S=k\ln W$ (термодинамическая вероятность, не ядро~$\mathcal{B}$)
и основным состоянием с $W=1$ при $T=0$.
На~$M$ физический вакуум не есть $|z|=0$: при $z=0$ фазовый поворот $z e^{i\Phi}$ обнуляет поле
и останавливает эволюцию; минимум «потенциала'' --- не в $z=0$.
При $|z|\to 0$ фазовый вращатель~$\Phi$ неограниченно возрастает --- вакуум --- \emph{осциллирующий} фон
с $\rho_{\mathrm{field}}>0$, не статический ноль.
\end{axiom}

\begin{axiom}[A6 · Второй закон (локально)]
\label{ax:A6}
На каждой $\varepsilon$-окрестности локальная фазовая энтропия (при фиксированной норме $N$)
не убывает: разрешено перемешивание / перераспределение фаз, запрещена «заморозка'' всех фаз в rest.
Рост энтропии не отменяет A3 --- не означает $N\to 0$.
\end{axiom}

\begin{axiom}[A7 · Планковский потолок плотности]
\label{ax:A7}
Энергетическая плотность ограничена планковским потолком $u_P$:
\[
  \rho_E(x,t)=\mu_P c^2 |z(x,t)|^2 \le u_P
  \quad\Leftrightarrow\quad
  |z|^2 \le 1 \ \text{(natural)}.
\]
Насыщающий вращатель использует знаменатель $+\varepsilon$, $\varepsilon\leftrightarrow u_P$.
\end{axiom}

\begin{axiom}[A8 · Макро-линейность]
\label{ax:A8}
При больших $|z|$ нелинейность вращателя затухает --- на $\Tlayer$ поле становится близким к линейному
(гладкость макроосреднение).
\end{axiom}

\section{Аналитичность и топология}

\begin{axiom}[A9 · Дискретная аналитичность (Коши--Риман)]
\label{ax:A9}
$g$ --- оператор удержания дискретной голоморфности: голономия $\zeta=(\sum_N z)\cdot z^*$
вместо ложных скачков $\Arg(\sum z)-\Arg z$.
CR-единственность на связном $\Lambda$ задаёт глобальное поле (риманова поверхность $\Lambda\times S^1$).
\end{axiom}

\begin{axiom}[A10 · Топологическая квантованность заряда]
\label{ax:A10}
Устойчивый дефект --- полюс на $\partial(\hv)$. Топологический заряд
\[
  n=\frac{1}{2\pi}\oint_{\partial(\hv)} d\arg z \in \mathbb{Z}.
\]
\end{axiom}

\begin{axiom}[A11 · Устойчивость солитона]
\label{ax:A11}
Размазанное (не голоморфное) состояние не устойчиво: вращатель с $K_P$ и $\Delta\phi_{\min}$
выталкивает полюс --- самофокусировка vortex, не smear.
\end{axiom}

\section{Изотропия, обратимость, симметрии}

\begin{axiom}[A12 · Изотropия / Lorentz на $\Tlayer$]
\label{ax:A12}
Микрошаг изотропен (равные веса на $N$). Макро-изотropный «свет'' ---
осреднение $\kgeom=1/\sqrt{2}$ на FCC (\cref{ch:carrier}) плюс биномиальное осреднение (\cref{ch:macro}).
\end{axiom}

\begin{axiom}[A13 · Обратимость микрошага]
\label{ax:A13}
$g$ обратим на дискретном алфавите: $\exists g^{-1}$, $g^{-1}(g(\Psi))=\Psi$.
Схема «лягушка'' второго порядка на $\mathbb{Z}_{N_{\mathrm{ring}}}[i]$:
\[
  Z^+(x,t)+Z^-(x,t)=2Z(x,t)+\lfloor\mathcal{N}(x,t)\rfloor.
\]
\end{axiom}

\begin{axiom}[A14 · Дискретные симметрии P, C, T, U(1)$_{\mathrm{vac}}$]
\label{ax:A14}
$g$ уважает дискретные симметрии: пространственная инверсия $P$, зарядовое сопряжение $C$ ($z\to z^*$),
обратимое движение по времени $T$ (с перестановка истории «лягушка''), глобальная U(1)-фаза вакуума.
\end{axiom}

\begin{axiom}[A15 · Принцип наименьшего действия / геометрия]
\label{ax:A15}
Безразмерные константы связи --- следствия геометрии носителя, не подгоняемых параметров:
\[
  \kgeom=\frac{R_{\mathrm{in}}}{R_{\mathrm{out}}},\quad
  \gamma=\kappa_{\mathrm{link}}=\frac{1}{|N|},\quad
  \alpha^*=1+\frac{1}{4\pi}.
\]
\end{axiom}

\begin{axiom}[A16 · Фермион / запрет Паули]
\label{ax:A16}
Спин~$1/2$: $R(N_{\mathrm{ring}}/2)\mapsto -z$ (оборот $2\pi$ меняет знак; $4\pi$ --- тождство).
Два \emph{совпадающих} спинора в одном $\PlanckCell$ запрещены (Pauli exclusion на решётке).
Минимальное фазовое действие $\Delta\phi_{\min}=\tfrac12$ задаёт $s_0=\hbar/2$.
\end{axiom}

\section{Детерминизм и «квантовая случайность''}

\begin{proposition}[Детерминизм $\Mlayer$]
\label{prop:determinism}
При \Axref{ax:A2} закон $g$ строго детерминирован: нет генератора случайных чисел в фундаментальном шаге.
\end{proposition}

\begin{proof}
По \Axref{ax:A2} обновление на узле~$x$ задаётся локальной функцией
\[
  z(x,t+1)=g\bigl(\{z(y,t)\}_{y\in N(x)},\,Z_t,\,Z_{t-1}\bigr),
\]
где $Z_t,Z_{t-1}$ --- регистры схемы «лягушка'' (\Axref{ax:A13}).
Правая часть --- детерминированная функция конечного набора прошлых данных;
в определении~$g$ (\cref{sec:g-law}) нет стохастического входа, интеграла по мере
и «бросания костей''. Значит, траектория $\Psi^t$ однозначно восстанавливается
из $\Psi^0$ и правила~$g$.
\end{proof}

Вероятность и Born на $\Tlayer$ --- следствие динамического хаоса (A5, A6) и грубого прибора
(\cref{ch:macro}), не «кости'' в $g$.

\section{Теорема о невозможности тепловой смерти \texorpdfstring{$\Mlayer$}{M}}
\label{sec:theorem-23}

\begin{definition}[M-эволюция]
\label{def:state-evolution}
$\Psi^{t+1}=g(\Psi^t)$ на счётном $\Lambda$; один $\htick$ на тик; $g$ постоянен.
\end{definition}

\begin{definition}[Останов · D1]
\label{def:D1}
Существует $t$ без шага $g$, или $\htick\to\infty$, или $g$ меняется / обнуляется.
\end{definition}

\begin{definition}[M heat death · D2]
\label{def:D2}
$\sum_x|z|^2\to 0$, или глобально $z\equiv 0$ устойчиво, или $g(\Psi^*)=\Psi^*$ с «мёртвым'' вакуумом без A5-кипения.
\end{definition}

\begin{definition}[$\Tlayer$ heat death · D3]
\label{def:D3}
Макроосреднение $\Phi=\mathcal{B}z$ теряет структуру ($\mathrm{Var}(\Phi)\to 0$) при $N\approx\mathrm{const}$ на $\Mlayer$.
\end{definition}

\begin{lemma}[Нет останов]
\label{lem:231}
При \cref{def:state-evolution} для всех $t\in\mathbb{Z}$: $\Psi^{t+1}=g(\Psi^t)$, $\htick$ и $g$ постоянны.
\end{lemma}

\begin{proof}
По \cref{def:state-evolution} на каждом такте выполняется $\Psi^{t+1}=g(\Psi^t)$,
шаг~$\htick$ и закон~$g$ не зависят от~$t$.
Остановка D1 требует либо прекращения применения~$g$, либо $\htick\to\infty$,
либо смены/обнуления~$g$ --- это изменение постановки задачи,
а не следствие эволюции при фиксированном локальном правиле.
\end{proof}

\begin{lemma}[Сохранение информации]
\label{lem:232}
$N(t+1)=N(t)$ для всех $t$.
\end{lemma}

\begin{proof}
Норма $N(t)=\sum_x z^\dagger(x,t)z(x,t)$ (\cref{def:field-a3}).
По \Axref{ax:A3} линейная часть~$g$ --- произведение локальных унитарных операторов
на рёбрах~$N(x)$, каждый сохраняет $z^\dagger z$ на затронутых узлах;
нелинейная часть --- только $z\mapsto z\,e^{i\Phi(z)}$, также сохраняющая $|z|$.
Композиция шага не меняет сумму $\sum_x|z|^2$, поэтому $N(t+1)=N(t)$.
\end{proof}

\begin{corollary}
\label{cor:232a}
$N\to 0$ на $\Mlayer$ невозможно, если $N(0)>0$.
\end{corollary}

\begin{proof}
Если $N(0)>0$, то индукция по \cref{lem:232} даёт $N(t)=N(0)>0$ для каждого~$t\in\mathbb{Z}$.
Последовательность $(N(t))$ не может сходиться к нулю при положительном начальном значении.
\end{proof}

\begin{lemma}[A6 не отменяет A3]
\label{lem:233}
A6 не влечёт $N\to 0$.
\end{lemma}

\begin{proof}
\Axref{ax:A6} разрешает лишь перераспределение фаз при \emph{фиксированной} норме~$N$
(перемешивание аргументов~$z$ в $\varepsilon$-окрестности).
\Axref{ax:A3} запрещает диссипацию модуля $|z|$.
Рост фазовой энтропии не содержит механизма $N\to 0$:
это разные оси условий, а не взаимное отменение.
\end{proof}

\begin{lemma}[Нет устойчивого $z\equiv 0$]
\label{lem:234}
$z(x,t)\equiv 0$ не устойчивое состояние $g$.
\end{lemma}

\begin{proof}
При $z\equiv 0$ унитарная фаза $z e^{i\Phi}$ остаётся нулём при любой~$\Phi$:
локальный шаг не может вывести поле из «мёртвой'' точки --- эволюция останавливается.
\Axref{ax:A5} отвергает $|z|=0$ как физический вакуум: требуется осциллирующий фон
с $\rho_{\mathrm{field}}>0$.
Следовательно, $z\equiv 0$ не является A5-физичным состоянием~$g$.
Вместе с \cref{cor:232a} путь D2 через $N\to 0$ или глобальный $z\equiv 0$ закрыт.
\end{proof}

\begin{lemma}[Микрообратимость]
\label{lem:235}
$g$ обратима на $Z_N[i]$.
\end{lemma}

\begin{proof}
\Axref{ax:A13} постулирует $g^{-1}$ с $g^{-1}(g(\Psi))=\Psi$ на $\mathbb{Z}_{N_{\mathrm{ring}}}[i]$.
Обратимость реализуется алгеброй «лягушка'':
\[
  Z^+(x,t)+Z^-(x,t)=2Z(x,t)+\lfloor\mathcal{N}(x,t)\rfloor
\]
(\cref{sec:congruence}, CL-1) при учёте фазовых импульсов~$\Phi$.
Каждый микрошаг --- перестановка на конечном пространстве конфигураций.
\end{proof}

\begin{lemma}[$\mathcal{B}$ необратима]
\label{lem:236}
При $R\ge 1$ осреднение $\mathcal{B}$ не инъективен.
\end{lemma}

\begin{proof}
По \cref{def:soft-readout} $\Phi=\mathcal{B}z$ --- свёртка с ядром
$W=\tfrac14[1,2,1]^{\otimes 2R}$ (на срезе) или эквивалентным $3\times 3$-ядром.
Ядро $W$ не сосредоточено в одной точке и не равно $\delta$-функции,
значит, существуют $z\ne z'$ с $\mathcal{B}z=\mathcal{B}z'$
(добавление к $z$ осцилляции, ортогональной ядру на конечном $\Lambda$).
Отображение $z\mapsto\mathcal{B}z$ не инъективно.
\end{proof}

\begin{proposition}[D3 не нарушает $\Mlayer$]
\label{prop:237}
$\nu_{\mathrm{CA}}$-затухание на $\Tlayer$ --- не член $g$; $N$ invariant (\cref{lem:232}).
\end{proposition}

\begin{proof}
\emph{Первое утверждение.}
Закон~$g$ на~$\Mlayer$ (\cref{sec:g-law}) --- композиция локальных унитарных преобразований
и фаз $e^{i\Phi(z)}$; в нём нет вещественного члена $\nu_{\mathrm{CA}}\nabla^2 z$,
 иначе $|z|$ перестал бы быть инвариантом в нарушение \Axref{ax:A3}.
Кинематическая вязкость $\nu_{\mathrm{CA}}\nabla^2\Phi$ появляется только в эффективном
уравнении для $\Phi=\mathcal{B}z$ на~$\Tlayer$ (\cref{eq:nlse-gl}, \cref{ch:macro}):
это остаток дискретного осреднения при конечных $\hl$, $\htick$,
а не новый оператор в~$g$.

\smallskip
\emph{Второе утверждение.}
Макроскопическое «затухание'' --- потеря различимости после необратимого~$\mathcal{B}$
(\cref{lem:236}): прибор не видит часть микрофазы.
На~$\Mlayer$ эта энергия не исчезает из~$N(t)$ (\cref{lem:232}):
по \Axref{ax:A5} она уходит в микро-«кипение'' --- перераспределение фаз
при $|z|\ge z_{\min}>0$; по \Axref{ax:A6} --- в рост локальной фазовой энтропии
при фиксированном~$N$.
Сценарий D3 ($\mathrm{Var}(\Phi)\to 0$ при $N\approx\mathrm{const}$) описывает
грубый readout на~$\Tlayer$, а не обнуление информации на~$\Mlayer$.
\end{proof}

\begin{definition}[Frozen tick · D4]
$\lfloor\mathcal{N}(\Psi)\rfloor\equiv 0$ на всём $\Lambda$.
\end{definition}

\begin{definition}[Глобальная неподвижная точка · D5]
$g(\Psi^*)=\Psi^*$ при $(Z_t,Z_{t-1})=(Z^*,Z^*)$ $\Leftrightarrow$ D4.
\end{definition}

\begin{lemma}[Классификация D5]
\label{lem:238a}
D5 $\Leftrightarrow$ $z(x,t)\equiv c$ на торе $\Lambda$, $c\in Z_N[i]$.
\end{lemma}

\begin{proof}
\emph{($\Rightarrow$)}~D5 означает $\lfloor\mathcal{N}\rfloor\equiv 0$ на всём $\Lambda$.
В симметричной «лягушке'' это даёт $\Phi=0$ на всех столкновениях,
дискретная CR-совместимость (\Axref{ax:A9}) и связность графа~$N$ $\Rightarrow$
$z(x)=\langle z\rangle_N$ на каждом узле, то есть $z(x)\equiv c$ для некоторого $c\in Z_N[i]$.

\smallskip
\emph{($\Leftarrow$)}~Если $z(x)\equiv c$, то все локальные разности и $\lfloor\mathcal{N}\rfloor$ обнуляются,
$(Z_t,Z_{t-1})$ не меняются и $g(\Psi^*)=\Psi^*$ --- неподвижная точка «лягушка''.
\end{proof}

\begin{lemma}[A5/A6 исключают физические D5]
\label{lem:238b}
Ни одна конфигурация из \cref{lem:238a} не A5-физична.
\end{lemma}

\begin{proof}
Пусть $z\equiv c$ --- неподвижная точка из \cref{lem:238a}.
При $c=0$ конфигурация противоречит \Axref{ax:A5} (мёртвый вакуум без кипения).
При $c\ne 0$ и отсутствии фазовой динамики поле статично на всех тактах:
\Axref{ax:A5} требует осциллирующего фона, \Axref{ax:A6} --- запрещает глобально
«замороженную'' голоморфность при фиксированном~$N$.
Ни один случай не A5-физичен.
\end{proof}

\begin{theorem}[Нет физических неподвижных точек без кипения]
\label{thm:238}
Класс A5-физических глобальные неподвижные точки $g(\Psi^*)=\Psi^*$ без микрокипения --- пуст.
\end{theorem}

\begin{proof}
Пусть $g(\Psi^*)=\Psi^*$ --- A5-физичная неподвижная точка без микрокипения.
По \cref{lem:238a} тогда $z(x)\equiv c$ на связном~$\Lambda$.
\cref{lem:238b} исключает $c=0$ (мёртвый вакуум) и $c\ne 0$ (статичная конфигурация
при требуемом A5-кипении и запрете A6 на «заморозку'').
Таких точек нет.
\end{proof}

\begin{theorem}[Тепловая смерть $\Mlayer$ невозможна]
\label{thm:main-23}
При \Axref{ax:A3}, \Axref{ax:A5}, \Axref{ax:A6}, \Axref{ax:A13} и законе~$g$
(гл.~\ref{ch:evolution}, \cref{sec:g-law}):
\begin{enumerate}[label=(\roman*)]
\item $\neg$D1 --- \cref{lem:231};
\item $\neg$D2 --- \cref{cor:232a}, \cref{lem:234} и \cref{thm:238};
\item A6 и A3 согласованы --- \cref{lem:233};
\item микроуровень обратим, макроуровень необратим --- \cref{lem:235}, \cref{lem:236} и \cref{prop:237};
\item нет A5-физических D5 --- \cref{thm:238}.
\end{enumerate}
\end{theorem}

\begin{proof}
Работаем в постановке \cref{def:state-evolution}: один такт~$\htick$, постоянный локальный закон~$g$.

\smallskip
\emph{(i) $\neg$D1.}
По \cref{def:state-evolution} для каждого $t\in\mathbb{Z}$ выполняется $\Psi^{t+1}=g(\Psi^t)$,
шаг~$\htick$ и правило~$g$ не зависят от~$t$.
Ни прекращение шага, ни $\htick\to\infty$, ни смена/обнуление~$g$ не следуют из такой эволуции ---
это было бы изменением модели, а не траектории (\cref{lem:231}).

\smallskip
\emph{(ii) $\neg$D2.}
Определение~\ref{def:D2} перечисляет три ветви «тепловой смерти'' на~$\Mlayer$.
Первая, $\sum_x|z|^2=N\to 0$, противоречит \Axref{ax:A3}:
при $N(0)>0$ лемма о сохранении~$N$ (\cref{lem:232}) даёт $N(t)=N(0)>0$ на всех тактах
(\cref{cor:232a}), так что предел $N\to 0$ недостижим.
Вторая ветвь --- глобально устойчивый $z\equiv 0$.
При нулевом поле фазовый поворот $z e^{i\Phi}$ остаётся нулём, эволюция не выводит конфигурацию
из этой точки; \Axref{ax:A5} не принимает $|z|=0$ как физический вакуум
(\cref{lem:234}).
Третья ветвь --- неподвижная точка $g(\Psi^*)=\Psi^*$ с «мёртвым'' вакуумом без микро-кипения A5.
Любая такая точка «лягушки'' есть однородное поле $z(x)\equiv c$ (\cref{lem:238a});
\cref{lem:238b} исключает $c=0$ (мёртвый вакуум) и статичное $c\ne 0$
(нет требуемого A5-осциллирующего фона и нарушен запрет A6 на глобальную «заморозку'' фаз).
Класс A5-физичных неподвижных точек без кипения пуст (\cref{thm:238}).
Все три ветви D2 закрыты.

\smallskip
\emph{(iii) Согласованность A6 и A3.}
\Axref{ax:A6} разрешает лишь перераспределение фаз при фиксированной норме~$N$;
\Axref{ax:A3} запрещает диссипацию $|z|$.
Рост фазовой энтропии не содержит механизма $N\to 0$ (\cref{lem:233}).

\smallskip
\emph{(iv) Микрообратимость и макронеобратимость.}
На~$\Mlayer$ шаг~$g$ обратим по \Axref{ax:A13}: существует $g^{-1}$ с $g^{-1}(g(\Psi))=\Psi$
на $\mathbb{Z}_{N_{\mathrm{ring}}}[i]$ (\cref{lem:235}).
Осреднение $\Phi=\mathcal{B}z$ при $R\ge 1$ не инъективно --- разные микросостояния дают одно макроосреднение
(\cref{lem:236}).
Кинематическая вязкость $\nu_{\mathrm{CA}}\nabla^2\Phi$ входит только в эффективное уравнение на~$\Tlayer$
(\cref{prop:237}), а не в~$g$; макроскопическое «затухание'' --- потеря различимости после~$\mathcal{B}$,
тогда как $N(t)$ на~$\Mlayer$ сохраняется и уходит в микро-перераспределение фаз (A5, A6).

\smallskip
\emph{(v) Нет A5-физических D5.}
Неподвижная точка D5 эквивалентна D4 (\cref{def:D5}) и сводится к $z\equiv c$ (\cref{lem:238a});
ни одна такая конфигурация не удовлетворяет \Axref{ax:A5} и \Axref{ax:A6} одновременно (\cref{lem:238b}),
откуда \cref{thm:238}.
\end{proof}
