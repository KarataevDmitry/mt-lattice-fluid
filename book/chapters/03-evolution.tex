\chapter{Закон эволюции}

\section{Общая структура}

Дискретность~$\Lambda$ и пара $(\hl,\htick)$ установлены теоремой~\ref{thm:discreteness}
и гл.~\ref{ch:carrier}.
Глобальное состояние эволюционирует по закону
\[
\Psi^t=\{z(x,t)\}_{x\in\Lambda}
\]
\[
  \Psi^{t+1}=g(\Psi^t),\qquad g=g_{\mathrm{nl}}\circ g_{\mathrm{lin}},
\]
где $g_{\mathrm{lin}}$ --- линейное смешивание по соседям~$N(x)$,
$g_{\mathrm{nl}}$ --- умножение на фазу $e^{i\Phi(z)}$.

\begin{definition}[Дискретный лапласиан]
$(\Delta_4 z)(x)=\sum_{y\in N(x)}(z(y)-z(x))$.
\end{definition}

На этом каркасе ещё не фиксированы $\gamma$, форма~$\Phi$, режим~$L$,
$\alpha^*$, $\varepsilon$ --- они определяются замыканием в \cref{sec:g-law}
и гл.~\ref{ch:si-sm}.

\section{Следствия абсолютных условий}

Условие~$A_3$ запрещает рост $\sum|z|^2$: допустимы лишь локальные унитарные преобразования
на связях и действительная фаза~$\Phi$.
$A_4$ ограничивает нелинейность множителем $e^{i\Phi}$.
$A_5$: при $|z|\to 0$ фазовый отклик~$\Phi$ неограниченно возрастает.
$A_6$ --- перемешивание при фиксированной норме.
$A_7$ --- насыщающий знаменатель с~$+\varepsilon$.
$A_8$ --- затухание нелинейности при больших $|z|$.
$A_9$ --- метрика голономия через инвариант~$\zeta$.

Линейное выражение $z+\gamma\Delta_4 z$ не унитарно;
допустимо произведение локальных унитарных преобразований (\cref{sec:g-law}).

\section{Фазовый инвариант \texorpdfstring{$\zeta$}{zeta}}

Разность $\Arg(\sum z/z)$ и функция $\atan2$ дают на решётке ложный скачок $2\pi$.
Физически значим величина
\[
  \zeta=(\textstyle\sum_N z)\cdot z^*,\qquad \Delta\phi=\wrap(\Arg\zeta),
\]
служащая входом насыщающего фазового импульса (\cref{sec:g-law}).

\section{Спин и группа \texorpdfstring{$SU(2)$}{SU(2)}}

\begin{definition}[Спинорное поле]
$z(x,t)=(z_1,z_2)^\top\in\mathbb{C}^2$, $|z|^2=z^\dagger z$.
\end{definition}

Глобальная $U(1)$-фаза $z\to z e^{i\Phi}$ описывает сохранение заряда,
но недостаточна для фермиона: оборот на $2\pi$ должен менять знак состояния,
на $4\pi$ --- возвращать его.

\begin{definition}[Дискретный вращатель]
\label{def:rotation-gate}
$N_{\mathrm{ring}}=512$, $\omega=e^{i2\pi/N_{\mathrm{ring}}}$, $\Phi\in\mathbb{Z}_{N_{\mathrm{ring}}}$:
\[
  R(\Phi):\ (U,V)\mapsto (U c_\Phi - V s_\Phi,\ U s_\Phi + V c_\Phi)\pmod N,
\]
где $(c_\Phi,s_\Phi)$ --- дискретные cos/sin угла по модулю~$N_{\mathrm{ring}}$.
\end{definition}

\begin{corollary}[Спин $1/2$]
$R(N_{\mathrm{ring}}/2)\mapsto -z$ на первичном вихре $\Rightarrow$
$U(2\pi)=-\mathbb{1}$, $U(4\pi)=+\mathbb{1}$.
\end{corollary}

\begin{proof}
По \cref{def:rotation-gate} угол $\Phi=N_{\mathrm{ring}}/2$ соответствует повороту на~$\pi$
в дискретном $SU(2)$.
На спиноре $(z_1,z_2)$ это даёт $(-z_1,-z_2)=-z$, то есть $U(2\pi)=-\mathbb{1}$;
удвоение угла возвращает знак: $U(4\pi)=+\mathbb{1}$ --- содержание $A_{16}$.
\end{proof}

Запрет Паули: два параллельных спинора в одной ячейке~$\PlanckCell$
приводят к $z_A+z_B\to 0$; коэффициент~$K_P$ задаёт отталкивание при $d\le\hl$.

\section{$U(1)_{\mathrm{vac}}$ и хиральность}

Глобальная фаза $z\to z e^{i\theta}$; инварианты --- голономия~$\zeta$
и ось Блоха.
\[
n=z^\dagger\sigma z/|z|^2
\]
Проекторы $P_L z=(z_1,0)^\top$, $P_R z=(0,z_2)^\top$
независимы в фазе свободного течения и смешиваются при столкновении
через~$K_P$ и аргумент~$\Arg$.

\section{Замыкание \texorpdfstring{$g$}{g} на \texorpdfstring{$\mathbb{Z}_N[i]$}{Z\_N[i]}}
\label{sec:g-law}

\begin{theorem}[Замыкание закона эволюции]
\label{thm:g-law}
Закон~$g$ на $Z_{N_{\mathrm{ring}}}[i]$:
\begin{enumerate}
\item схема «лягушка'' второго порядка
\[
  Z(x,t+\Delta t)+Z(x,t-\Delta t)=2Z(x,t)+\lfloor\mathcal{N}\rfloor;
\]
\item действительная и мнимая части голономия из суммы соседей без $\atan2$;
\item насыщающий фазовый импульс
\[
  \Phi=\Bigl\lfloor\frac{K_P\zeta_{\mathrm{imag}}}{\zeta_{\mathrm{real}}+|Z|^2+K_P}\Bigr\rfloor,\quad
  |\Phi|\ge\Delta\phi_{\mathrm{disc}}\ \text{или}\ \Phi=0;
\]
\item $\lfloor\mathcal{N}\rfloor=R(\Phi)-Z$ на обеих компонентах; обратимость~$A_{13}$.
\end{enumerate}
\end{theorem}

\begin{proof}
Замыкание идёт по цепочке условий $A_3$--$A_{16}$; покажем, что каждый пункт теоремы
вынужден, а не подобран.

\smallskip
\emph{Схема «лягушка''.}
Условие~$A_3$ (\cref{ax:A3}) запрещает рост $\sum_x|z(x)|^2$ на такт.
Линейное обновление вида $z+\gamma\Delta_4 z$ не унитарно (§2 предыдущего раздела),
поэтому допустима дискретная волновая форма второго порядка по времени,
сохраняющая «энергию связи'' на ребре --- уравнение «лягушки''
\[
  Z(x,t+\Delta t)+Z(x,t-\Delta t)=2Z(x,t)+\lfloor\mathcal{N}(x,t)\rfloor.
\]
Условие~$A_{13}$ требует явного $g^{-1}$; та же формула обратима при известных
$Z(x,t\pm\Delta t)$ и $\lfloor\mathcal{N}\rfloor$.
Округление $\lfloor\cdot\rfloor$ --- работа в кольце $\mathbb{Z}_{N_{\mathrm{ring}}}[i]$,
а не произвольная нелинейность.

\smallskip
\emph{Голономия.}
Разность $\Arg(z(y)/z(x))$ на решётке скачет на $2\pi$ (\cref{sec:evolution}, §3).
$A_9$ (\cref{ax:A9}) фиксирует физически значимую фазу через
$\zeta=(\sum_{y\in N(x)} z(y))\cdot z(x)^*$ без $\atan2$ по парам
(§3 настоящей главы);
$\zeta_{\mathrm{real}}$, $\zeta_{\mathrm{imag}}$ --- действительная и мнимая части
этой голономии и входят в $\lfloor\mathcal{N}\rfloor$.

\smallskip
\emph{Насыщающий импульс.}
$A_5$ (\cref{ax:A5}): при $|z|\to 0$ знаменатель saturating gate обращается в ноль,
фазовый отклик $\Phi$ неограничен --- вакуум не может быть статическим $z=0$.
$A_7$ (\cref{ax:A7}): при больших $|z|$ рост ограничен добавкой $|Z|^2+K_P$ в знаменатель.
Отсюда
\[
  \Phi=\Bigl\lfloor\frac{K_P\,\zeta_{\mathrm{imag}}}{\zeta_{\mathrm{real}}+|Z|^2+K_P}\Bigr\rfloor,
\]
с порогом $|\Phi|\ge\Delta\phi_{\mathrm{disc}}$ или $\Phi=0$ из~$A_{16}$ (\cref{ax:A16}).

\smallskip
\emph{Вращатель и обратимость.}
$A_4$ (\cref{ax:A4}) допускает нелинейность только как унитарную фазу на~$\mathbb{C}^2$;
реализуем её дискретным $SU(2)$-вращателем $R(\Phi)$ (\cref{def:rotation-gate}).
Подстановка $\lfloor\mathcal{N}\rfloor=R(\Phi)-Z$ в «лягушку'' замыкает шаг.
Эквивалентность $Z^++Z^-=2Z+\lfloor\mathcal{N}\rfloor\Leftrightarrow g^{-1}$
--- CL-1 (\cref{sec:congruence}), проверка mod~$N_{\mathrm{ring}}=512$.
Лестница CL-1--CL-8 согласует энергию, импульс, секторы~$N_\phi$ и спин~$1/2$
с тем же $N_{\mathrm{ring}}$ (\cref{sec:bit-budget}).
\end{proof}

\subsection{Информационный бюджет ячейки}
\label{sec:bit-budget}

\[
  B_{\hv}=\frac{2\pi}{\ln 2}\approx 9{,}065\ \text{бит},\quad
  N_{\mathrm{ring}}=2^{\lfloor B_{\hv}\rfloor}=512,\quad
  N_\phi=\lceil 4\pi\rceil=13,\quad
  \Delta\phi_{\mathrm{disc}}=41.
\]

\subsection{Лестница согласованности}
\label{sec:congruence}

\begin{proposition}[CL-1 -- CL-8]
По модулю~$N_{\mathrm{ring}}$:
\begin{itemize}
\item CL-1: $Z^++Z^-=2Z+\lfloor\mathcal{N}\rfloor$ $\Leftrightarrow$ $g^{-1}$ ($A_{13}$);
\item CL-2: $R(N/2)\mapsto -z$ (спин~$1/2$);
\item CL-3: $|\Phi|\ge\Delta\phi_{\mathrm{disc}}$ или $\Phi=0$ ($A_{16}$);
\item CL-4: $n_E=\lfloor|\Phi|/\Delta\phi_{\mathrm{disc}}\rfloor$, $E=n_E E_0$;
\item CL-5--CL-8: условия на greatest common divisor для генератора~$\mathbb{Z}_N$,
  секторов~$N_\phi$, учёта энергии и импульса.
\end{itemize}
\end{proposition}

\begin{proof}
CL-1 --- алгебраическая эквивалентность обратимости «лягушки'' (\cref{ax:A13}).
CL-2 следует из \cref{def:rotation-gate} и доказанного следствия о спине~$1/2$.
CL-3 --- порог $A_{16}$ в формуле~$\Phi$ (\cref{thm:g-law}).
CL-4 связывает $|\Phi|$ с $n_E$ и $E=n_E E_0$ (\cref{lem:51e}).
CL-5--CL-8 --- арифметика $\gcd$ для генератора $\mathbb{Z}_N$,
$N_\phi=\lceil 4\pi\rceil=13$ и дискретных законов сохранения
импульса/энергии (\cref{lem:51d}, \cref{sec:theorem-51}) при $N_{\mathrm{ring}}=512$,
$\Delta\phi_{\mathrm{disc}}=41$ из \cref{sec:bit-budget}.
\end{proof}

Топологический заряд $n\in\mathbb{Z}$ ($A_{10}$) не сводится к mod~$N_{\mathrm{ring}}$.
