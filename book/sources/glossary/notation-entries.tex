\newcommand{\NotationDescMlayer}{микроуровень: решётка, локальный закон g.}
\newcommand{\NotationDescTlayer}{макроуровень: осреднённая физика, SI, эксперимент.}
\newcommand{\NotationDescLambda}{счётное множество узлов носителя.}
\newcommand{\NotationDescvp}{планкон --- элементарная ячейка на M.}
\newcommand{\NotationDescg}{локальный обратимый закон эволюции; конфигурация в такт t.}
\newcommand{\NotationDesca}{элементарный шаг решётки (шаг--такт); в коде hL.}
\newcommand{\NotationDescdt}{длительность одного такта g (шаг--такт); в коде hT.}
\newcommand{\NotationDesctau}{номер такта (поколение CA); связь с t\_SI после SI-моста.}
\newcommand{\NotationDescCzero}{тактовая скорость сигнала на M; c\_0=a/\(\Delta t\) (шаг--такт).}
\newcommand{\NotationDescCspeed}{макроскопическая скорость света на макроуровне (SI-мост).}
\newcommand{\NotationDesckgeom}{безразмерный коэффициент геометрии носителя (kappa в коде).}
\newcommand{\NotationDescdhdt}{внешние координаты dh и dt (dh=a, dt=\(\Delta t\)).}
\newcommand{\NotationDescszero}{ступень действия на одну ячейку и один такт; порождает лестницу квантов.}
\newcommand{\NotationDescpzero}{импульсный квант: p\_0=s\_0/dh.}
\newcommand{\NotationDescEzero}{энергетический квант: E\_0=s\_0/dt.}
\newcommand{\NotationDescFzero}{силовой квант: F\_0=s\_0/(dh·dt).}
\newcommand{\NotationDescmzero}{массовый квант лестницы: m\_0=E\_0/c\_0\textasciicircum 2; SI-мост.}
\newcommand{\NotationDescL}{длина в метрах (слой T).}
\newcommand{\NotationDesclambda}{характерная длина на шаге шкалы (комптоновская, боровская, \ldots).}
\newcommand{\NotationDesclambdaC}{комптоновская длина электрона.}
\newcommand{\NotationDesclambdak}{шаги IR-лестницы.}
\newcommand{\NotationDescellP}{планковская длина; на носителе \(\ell_P=a\) (SI-мост).}
\newcommand{\NotationDescNc}{безразмерное \(\lambda_C/\ell_P\); не «голое» N.}
\newcommand{\NotationDescloglambda}{порядок длины в таблице шкалы.}
\newcommand{\NotationDesctSI}{время в секундах (слой T).}
\newcommand{\NotationDesctP}{планковское время; \(\Delta t=\kappa\,t_P\) (SI-мост).}
\newcommand{\NotationDesclogtau}{порядок числа тактов (космология; verify: log10\_N\_today).}
\newcommand{\NotationDescNring}{размер фазового кольца N\_ring (дискретная фаза).}
\newcommand{\NotationDescBhv}{информационный бюджет ячейки; связь с \(N_{\mathrm{ring}}\) (дискретная фаза).}
\newcommand{\NotationDescphiPhi}{дискретная фаза $\phaseVarphi_{\mathrm{disc}}$ и импульс за тик $\symCapitalPhi$.}
\newcommand{\NotationDescNphi}{число секторов Heisenberg-кванта фазы.}
\newcommand{\NotationDescNFCC}{число ближайших соседей FCC; степени \(N_{12}^m\) --- геометрия звезды.}
\newcommand{\NotationDescNpack}{число упаковки атомной ячейки (открытый вывод).}
\newcommand{\NotationDescNhier}{число ступеней ослабления на T (VEV).}
