% --- Печать по блокам (заголовок + отдельная longtable на блок) ---
\gdef\notationBlkMlist{}
\gdef\notationBlkTlist{}
\gdef\notationBlkMtitle{}
\gdef\notationBlkMkeys{}
\gdef\notationBlkTtitle{}
\gdef\notationBlkTkeys{}

\newcommand{\notationBlkMaddkey}[1]{%
  \ifdefvoid{\notationBlkMkeys}{%
    \gdef\notationBlkMkeys{#1}%
  }{%
    \edef\notationBlkMkeys{\notationBlkMkeys,#1}%
  }%
}
\newcommand{\notationBlkTaddkey}[1]{%
  \ifdefvoid{\notationBlkTkeys}{%
    \gdef\notationBlkTkeys{#1}%
  }{%
    \edef\notationBlkTkeys{\notationBlkTkeys,#1}%
  }%
}
\newcommand{\notationBlkMcloseblock}{%
  \ifdefvoid{\notationBlkMkeys}{}{%
    \appto\notationBlkMlist{%
      \notationBlkprint{M}{\notationBlkMtitle}{\notationBlkMkeys}%
    }%
    \gdef\notationBlkMkeys{}%
  }%
}
\newcommand{\notationBlkTcloseblock}{%
  \ifdefvoid{\notationBlkTkeys}{}{%
    \appto\notationBlkTlist{%
      \notationBlkprint{T}{\notationBlkTtitle}{\notationBlkTkeys}%
    }%
    \gdef\notationBlkTkeys{}%
  }%
}
\newcommand{\notationBlkprint}[3]{%
  \Needspace{9\baselineskip}%
  \noindent #2\par
  \vspace{0.3em}%
  \ifstrequal{#1}{M}{%
    \printunsrtglossary[type=notationM,style=notationlong,nonumberlist,sort=use]{#3}%
  }{%
    \printunsrtglossary[type=notationT,style=notationlong,nonumberlist,sort=use]{#3}%
  }%
}

\newcommand{\notationlongtablecols}{@{}p{0.17\linewidth}p{0.53\linewidth}p{0.24\linewidth}@{}}
\newcommand{\notationtablehead}{%
  \toprule
  \foreignlanguage{russian}{Обозначение} &
  \foreignlanguage{russian}{Смысл} &
  \foreignlanguage{russian}{Единица} \\
  \midrule
}
\newglossarystyle{notationlong}{%
  \renewenvironment{theglossary}{%
    \small
    \setlength{\tabcolsep}{5pt}%
    \renewcommand{\arraystretch}{1.12}%
    \begin{longtable}{\notationlongtablecols}%
    \notationtablehead
    \endfirsthead
    \notationtablehead
    \endhead
    \bottomrule
    \endfoot
  }{%
    \end{longtable}%
  }%
  \renewcommand*{\glossaryheader}{}%
  \renewcommand*{\glsgroupheading}[1]{}%
  \renewcommand{\glossentry}[2]{%
    \glstarget{##1}{\glossentryname{##1}} &
    \glossentrydesc{##1}\glspostdescription\space ##2 &
    \notationentryunit{##1}\tabularnewline
  }%
  \renewcommand{\subglossentry}[3]{}%
  \renewcommand*{\glsgroupskip}{}%
}

\newcommand{\printnotationglossary}[1]{%
  \begingroup
  \setglossarysection{none}%
  \ifstrequal{#1}{M}{%
    \notationBlkMlist
  }{%
    \notationBlkTlist
  }%
  \endgroup
}

\newcommand{\newnotationsectionM}[3]{%
  \notationBlkMcloseblock
  \gdef\notationBlkMtitle{#3}%
  \newglossaryentry{#1}{%
    type=notationM, sort={#2},
    name={#3},
    description={}%
  }%
}

\newcommand{\newnotationsectionT}[3]{%
  \notationBlkTcloseblock
  \gdef\notationBlkTtitle{#3}%
  \newglossaryentry{#1}{%
    type=notationT, sort={#2},
    name={#3},
    description={}%
  }%
}

\newcommand{\newnotationsection}[3]{\newnotationsectionM{#1}{#2}{#3}}
