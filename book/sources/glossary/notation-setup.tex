\newcommand{\notationlongtablecols}{@{}p{0.17\linewidth}p{0.53\linewidth}p{0.24\linewidth}@{}}
\newcommand{\notationtablehead}{%
  \toprule
  \foreignlanguage{russian}{Обозначение} &
  \foreignlanguage{russian}{Смысл} &
  \foreignlanguage{russian}{Единица} \\
  \midrule
}
\newcommand{\notationtableheadcontinue}{%
  \midrule
  \foreignlanguage{russian}{Обозначение} &
  \foreignlanguage{russian}{Смысл} &
  \foreignlanguage{russian}{Единица} \\
  \midrule
}
% Если до конца страницы мало места — разрыв перед подсекцией (внутри longtable).
\newcommand{\notationpagebreakifneeded}{%
  \noalign{%
    \ifdim\pagetotal>\dimexpr\pagegoal-10\baselineskip\relax
      \pagebreak[3]%
    \fi
  }%
}

\newif\ifnotationtablehasbody
\newglossarystyle{notationlong}{%
  \renewenvironment{theglossary}{%
    \notationtablehasbodyfalse
    \small
    \setlength{\tabcolsep}{5pt}%
    \renewcommand{\arraystretch}{1.12}%
    \begin{longtable}{\notationlongtablecols}%
    \notationtablehead
    \endfirsthead
    \notationtableheadcontinue
    \endhead
    \bottomrule
    \endfoot
  }{%
    \end{longtable}%
  }%
  \renewcommand*{\glossaryheader}{}%
  \renewcommand*{\glsgroupheading}[1]{}%
  \renewcommand{\glossentry}[2]{%
    \ifglshasdesc{##1}{%
      \notationtablehasbodytrue
      \glstarget{##1}{\glossentryname{##1}} &
      \glossentrydesc{##1}\glspostdescription\space ##2 &
      \notationentryunit{##1}\tabularnewline
    }{%
      \ifnotationtablehasbody
        \notationpagebreakifneeded
        \noalign{\vspace{0.45em}}%
      \fi
      {\itshape\glossentryname{##1}} & & \tabularnewline
      \ifnotationtablehasbody
        \notationtableheadcontinue
      \fi
    }%
  }%
  \renewcommand{\subglossentry}[3]{}%
  \renewcommand*{\glsgroupskip}{}%
}

\newcommand{\printnotationglossary}[1]{%
  \begingroup
  \setglossarysection{none}%
  \ifstrequal{#1}{M}{%
    \printunsrtglossary[type=notationM,style=notationlong,sort=standard,nonumberlist]%
  }{%
    \printunsrtglossary[type=notationT,style=notationlong,sort=standard,nonumberlist]%
  }%
  \endgroup
}
