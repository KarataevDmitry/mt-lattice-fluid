\newcommand{\notationentryunit}[1]{\glsentryuseri{#1}}

\newcommand{\newnotationentryM}[5]{%
  \newglossaryentry{#1}{%
    type=notationM, sort={#2},
    name={#3},
    description={#4},
    user1={#5}%
  }%
}

\newcommand{\newnotationentryT}[5]{%
  \newglossaryentry{#1}{%
    type=notationT, sort={#2},
    name={#3},
    description={#4},
    user1={#5}%
  }%
}

% Совместимость (новые записи — явно M или T).

\newglossarystyle{notationlong}{%
  \renewenvironment{theglossary}{%
    \small
    \setlength{\tabcolsep}{5pt}%
    \renewcommand{\arraystretch}{1.12}%
    \begin{longtable}{@{}p{0.17\linewidth}p{0.53\linewidth}p{0.24\linewidth}@{}}%
    \toprule
    \foreignlanguage{russian}{Обозначение} &
    \foreignlanguage{russian}{Смысл} &
    \foreignlanguage{russian}{Единица} \\
    \midrule
    \endfirsthead
    \toprule
    \foreignlanguage{russian}{Обозначение} &
    \foreignlanguage{russian}{Смысл} &
    \foreignlanguage{russian}{Единица} \\
    \midrule
    \endhead
    \bottomrule
    \endfoot
  }{%
    \end{longtable}%
  }%
  \renewcommand*{\glossaryheader}{}%
  \renewcommand*{\glsgroupheading}[1]{}%
  \renewcommand{\glossentry}[2]{%
    \ifglshasdesc{##1}{%
      \glstarget{##1}{\glossentryname{##1}} &
      \glossentrydesc{##1}\glspostdescription\space ##2 &
      \notationentryunit{##1}\tabularnewline
    }{%
      \glstarget{##1}{\glossentryname{##1}} & & \tabularnewline
    }%
  }%
  \renewcommand{\subglossentry}[3]{}%
  \renewcommand*{\glsgroupskip}{}%
}

\newcommand{\printnotationglossary}[1]{%
  \begingroup
  \setglossarysection{none}%
  \ifstrequal{#1}{M}{%
    \printunsrtglossary[type=notationM,style=notationlong,sort=standard,nonumberlist]%
  }{%
    \printunsrtglossary[type=notationT,style=notationlong,sort=standard,nonumberlist]%
  }%
  \endgroup
}
