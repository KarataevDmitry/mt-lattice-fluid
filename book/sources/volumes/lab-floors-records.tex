\begin{labseries}{Серия CE-F: этаж~1, составные узлы и оболочки}{sec:lab-series-ce-f}

\begin{labrecord}{CE-F-01 · Двухтельное столкновение $p^+p^-$, $p^+p^+$, $p^-p^-$ на кипящем вакууме}{lab:ce-f-01}
\labgoal{Проверить §5.0.3: аннигиляция только в канале разных знаков; однознаковые пары не должны
  «гаснуть'' по тому же критерию.}
\labhypothesis{При ближнем столкновении на A5-boil пара $p^+p^-$ теряет норму/узел по порогу сценария;
  $p^+p^+$ и $p^-p^-$ --- нет (контроль).}
\labmodel{MODEL~\S5.0.3; этаж~1, динамика на кипящем вакууме.}
\labsetup{Два планкон-узла на FCC-решётке, NN-столкновение; tick-resolve по полному шагу $g$;
  начальное состояние --- вихрь на boil; размер куба достаточен для изоляции.}
\labobservables{Флаг аннигиляции; остаточная норма; число тактов до события.}
\labcriterion{PASS: pm аннигилирует; pp и mm --- нет; воспроизводимость при смене чётности центра.}
\labanalysis{Тип~A: несколько seeds; тип~B: порог --- свойство сценария, не PDG.}
\labconclusion{Канал аннигиляции на этаже~1 подтверждён на минимальной двухтельной постановке;
  перенос на $n$-тело --- CE-F-02.}
\labreproduce{Реестр: \texttt{Floor1\_two\_body\_nn}.}
\end{labrecord}

\begin{labrecord}{CE-F-02 · Цепочка из четырёх узлов (FCC-альтернация знаков)}{lab:ce-f-02}
\labgoal{Устойчивость аннигиляционной логики при $n>2$ без раннего обрыва прогона в не-pm каналах.}
\labhypothesis{Чередующаяся цепочка даёт каскад pm-событий; однородный знак --- контроль без ложного гашения.}
\labmodel{MODEL~\S5.0.3; $n$-телесная постановка на том же аппарате.}
\labsetup{Четыре узла вдоль NN-цепи; варианты \texttt{chain4\_alt\_pm}, \texttt{cluster\_ppmm}, \texttt{chain4\_pp}.}
\labobservables{Число аннигиляций; финальная норма; wall-time (диагностика, не критерий физики).}
\labcriterion{Согласованность с CE-F-01 на pm-подцепях; pp-контроль не аннигилирует.}
\labanalysis{Серия CE-F: один аппарат, разные IC; выводы независимы от одного seed.}
\labconclusion{Минимальный $n$-body dogfood закрыт; спектроскопия bound-state --- открыта.}
\labreproduce{Реестр: \texttt{Floor1\_n\_body\_chain}.}
\end{labrecord}

\labgoal{Связать аннигиляцию на решётке с прокси двухфронтовой структуры на средней плоскости.}
\labhypothesis{После pm на boil coarse-поле даёт $\ge 2$ пика на $\Delta|\Phi|$ и ось диполя, согласованную с инструментом T-ансамбля.}

\begin{labrecord}{CE-A-02 · Head-on ансамбль: ось, mod-$\pi$, отделение $\tau_M$ от PDG}{lab:ce-a-02}
\labgoal{Статистика T2-прокси: back-to-back, отслеживание оси инъекции; $\tau_M=n_{\mathrm{ticks}}\cdot hT$ не тождественно para-Ps.}
\labhypothesis{Ансамбль вращений даёт стабильную ось (mod $\pi$); $\tau_M/\tau_{\mathrm{PDG}}\ll 1$ (разные часы).}
\labmodel{MODEL~\S5.0.3; \cref{ch:measurement-processing} для ансамбля.}
\labsetup{Гауссовы пакеты head-on; ensemble $\ge 8$; размер/шаги сценария фиксированы в реестре.}
\labobservables{$\tau_M$; ошибка оси к инъекции; peaks on $\delta$; CV гистограммы угла (диагностика).}
\labcriterion{PASS: back-to-back proxy, axis tracks $\pi$, collision clock separate; для среднего угла $\bar\theta$ при $n$ прогонах
  интервал $\bar\theta\pm t_{n-1,0.975}\,s/\sqrt{n}$ (\cref{sec:student}), не замена $\pm s$.}
\labanalysis{Тип~A: типичный прогон реестра ($n=12$): $\bar\varepsilon_{\pi}\approx 6\times 10^{-2}\,\mathrm{rad}$

  Полуинтервал $t\,s/\sqrt{n}$ порядка $10^{-2}\,\mathrm{rad}$ --- статистика прокси, не PDG.
  PDG $\tau$ --- только CE-A-03.}
\labconclusion{T-статистика диполя закрыта на классе head-on; bound-state lattice $\tau$ --- open.}
\labreproduce{\texttt{Annihilation\_T\_stats}, bundle \texttt{Annihilation\_M\_T\_bundle}.}
\end{labrecord}

\begin{labrecord}{CE-A-03 · Para-Ps $\tau$: QED, лестница, strict upstream}{lab:ce-a-03}
\labgoal{Сопоставить $\tau_{^1S_0}$ с PDG без подмены head-on $\tau_M$; strict --- без кругового CODATA $m_e$.}
\labhypothesis{$\tau\approx 2\hbar/(m_e c^2\alpha^5)$ и лестница $N_c/\alpha^5$ совпадают; upstream $m_e$ даёт больший $E_n$, чем CODATA-path.}
\labmodel{MODEL~\S5.0.3; \eqref{eq:para-ps-tau}--\eqref{eq:tau-relative-unc}.}
\labsetup{Массы §8.2 upstream; $\alpha_{\mathrm{preferred}}$; отдельно CODATA-path как диагностика; ortho/para ratio leading.}
\labobservables{$\tau_{\mathrm{mod}}$; $E_n$ к PDG; ratio $\tau_{\mathrm{ortho}}/\tau_{\mathrm{para}}$; $\tau_M/\tau_{\mathrm{PDG}}$.}
\labcriterion{Strict: $|\tau_{\mathrm{up}}-\tau_{\mathrm{PDG}}|/\tau_{\mathrm{PDG}}$ в полосе от $m_e$ §8.2;
  anti-circular; $E_n$ upstream $\lesssim 2{,}5$ при GUM-переносе; bridge (CODATA $m_e$) --- диагностика, не единственный gate.}
\labanalysis{%
\textbf{Эталон:} $\tau_{\mathrm{PDG}}=125{,}087\,\mathrm{ps}$, $u(\tau)\approx 0{,}027\,\mathrm{ps}$
($u/\tau\approx 2{,}2\times 10^{-4}$).
\textbf{Косвенные модели} (ведущий QED, \eqref{eq:para-ps-tau}):
$\tau_{\mathrm{up}}\approx 123{,}94\,\mathrm{ps}$ ($\delta/\tau\approx 9{,}2\times 10^{-3}$, upstream $m_e$);
$\tau_{\mathrm{CODATA}\,m_e}\approx 124{,}49\,\mathrm{ps}$ ($\delta/\tau\approx 4{,}7\times 10^{-3}$).
\textbf{GUM} (\eqref{eq:tau-relative-unc}): для upstream доминирует $u(m_e)/m_e\approx 4{,}5\times 10^{-3}$
(§8.2 vs PDG-электрон), $5\,u(\alpha)/\alpha$ из CODATA --- второй порядок;
$u_{\mathrm{mod}}/\tau\approx 4{,}5\times 10^{-3}$.
Тогда $u_{\mathrm{comb}}/\tau\approx\sqrt{(4{,}5\times 10^{-3})^2+(2{,}2\times 10^{-4})^2}$
и $E_n^{\mathrm{(up)}}\approx 2{,}0$ (\eqref{eq:en-score}) --- согласие в пределах $\sim 2\sigma$ по заявленным $u$,
без подгонки $\alpha^5$.
\textbf{CODATA-path:} при $u(m_e)$ таблицы $\sim 10^{-10}$ остаток $4{,}7\times 10^{-3}$ даёт $E_n\gg 1$:
это не «точное попадание'', а \emph{систематика типа~B} (ведущий порядок без полной $\Gamma_{\mathrm{QED}}$),
поэтому strict запрещает считать CODATA-$m_e$ единственным входом.
\textbf{Ortho/para:} отношение $\tau_{\mathrm{ortho}}/\tau_{\mathrm{para}}$ leading $\approx 8{,}5\times 10^2$
vs PDG $\approx 1{,}14\times 10^3$ ($\delta_{\mathrm{rel}}\approx 26\%$) --- тот же класс типа~B, не $E_n$ к узкому $u$.
Числа воспроизводимы: \texttt{Annihilation\_PDG\_tau\_strict} (поля \texttt{E\_n\_upstream}, \texttt{u\_tau\_*}).}
\labconclusion{Алгебраический мост T-door закрыт в полосе upstream-$E_n$; sim-bound $n_{\mathrm{ticks}}\to 125\,$ps --- открытый опыт.}
\labstatus{Закрыт (алгебра strict); lattice $\tau$ --- открыт.}
\labreproduce{\texttt{Annihilation\_PDG\_tau\_strict}, \texttt{Annihilation\_PDG\_tau\_bridge}.}
\end{labrecord}
